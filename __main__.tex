\begin{frame}{Background}
\begin{exampleblock}{}
\centering
    \textbf{Problem}: How to learn a target probability distribution $\pi$ on $\R^d$.
\end{exampleblock}
\begin{itemize}
    \item Sampling (e.g: $\pi\propto\exp(-V)$ is the posterior distribution in Bayesian inference). 
    \item Generative models (e.g: $\pi$ is the distribution of an image dataset). 
\end{itemize}
\end{frame}


\begin{frame}{Background: Optimization in the space of probability measures}
    \begin{exampleblock}{}
    \centering
        \textbf{Problem}: How to learn a target probability distribution $\pi$ on $\R^d$.
    \end{exampleblock}
    \begin{itemize}
        \item This problem can be written as an \blue{optimization} problem on $\calP_2(\R^d)$.
        \begin{talign*}
            \arg\min_{\mu \in \calP_2(\R^d)} D(\mu \mid \pi) . 
        \end{talign*}
        \item Here $D$ is a similarity metric or distance, e.g. Kullback–Leibler divergence. 
        \begin{itemize}
            \item $D(\mu | \pi) = 0$ if and only if $\mu=\pi$. 
        \end{itemize}
        \item \red{$\calP_2(\R^d)$ denotes the space of probability measures with a finite second moment}. 
        \begin{itemize}
            \item We primarily consider $\calP_2(\R^d)$ rather than $\calP(\R^d)$ for the nice geometrical properties. 
        \end{itemize}
        \item How to find the minimum? Gradient descent!
    \end{itemize}
\end{frame}


\begin{frame}{Background: Euclidean gradient flow}
    \begin{itemize}
        \item Euclidean gradient flow of an objective $F:\R^d \to \R$ 
        \begin{talign*}
            \partial_t x_t = -v(x_t), \quad v = \nabla F .
        \end{talign*}
        \item $\nabla F$ denotes the gradient of $F$. 
        \item This is the continuous-time analogue of gradient descent: \begin{talign*}
            x_{n+1} = x_n - \gamma v(x_n),
        \end{talign*} 
        where $\gamma > 0$ is the step size. 
        \item When $F$ is both \green{strongly convex and smooth}, Euclidean gradient \orange{descent} converges exponentially fast~\citep{boyd2004convex}. 
    \end{itemize}
\end{frame}



\begin{frame}{Background: Wasserstein gradient flow}
    \begin{exampleblock}{}
    \centering
        \textbf{Challenge}: How to find $\arg\min_{\mu \in \calP_2(\R^d)} D(\mu \mid \pi)$?
    \end{exampleblock}
    \begin{itemize}
        \item Gradient descent! Wait... How do we define gradients in $\calP_2(\R^d)$? 
        \item Endow $\calP_2(\R^d)$ with the Wasserstein-2 distance $W_2$.
        \begin{talign*}
            \red{W_2^2(\nu, \mu)} = \smallint \| T(x) - x\|^2 \dd \nu(x) = \| T - \Id\|_{L^2(\nu)}^2 . 
        \end{talign*}
        \vspace{-10pt}
        \begin{itemize}
            \item $W_2^2(\nu, \mu)$ means the \orange{minimal energy} takes to transport mass from $\nu$ to $\mu$. 
            \item $T:\R^d\to\R^d$ is the \red{optimal transport map} from $\nu$ to $\mu$. 
        \end{itemize}
        \item $(\calP_2, W_2)$ can be 'treated` as a Riemann manifold under Otto's calculus~\citep{otto2001geometry}. 
        \item The \green{tangent space} $\calT_\mu\calP_2(\R^d)$ at $\mu\in\calP_2(\R^d)$ is a dense subset of $L^2(\mu)$. 
    \end{itemize}

            clip
        ]{figures/ot_wasserstein_gradient_illustration.pdf}

\end{frame}

\begin{defi}[Wasserstein gradient]
Let $\calF: \calP_2(\R^d) \rightarrow \R$.
The \red{Wasserstein gradient of $\calF$} evaluated at $\mu \in \calP_2(\R^d)$ is the unique function \blue{$\nabla_{W_2} \calF(\mu): \R^d \rightarrow \R^d$}, s.t. for any $T\in \calT_\mu\calP_2(\R^d)$,
\begin{align*}
\lim _{\epsilon \rightarrow 0} \frac{1}{\epsilon}\left[\calF\left( (\Id+\epsilon T)_\# \mu \right) - \calF(\mu)\right] = \left\langle \nabla_{W_2} \calF, T \right\rangle_{L^2(\mu)} .
\end{align*}
\end{defi}
\begin{itemize}
    \item The gradient is defined along a `curve' $(\Id+\epsilon T)_\# \mu$ in $\calP_2(\R^d)$. 
\end{itemize}
\end{frame}

\begin{frame}{Background: Wasserstein gradient  flow}
\begin{defi}[Wasserstein gradient  flow]
\red{Wasserstein gradient flow} of $\calF:\calP_2(\R^d)\to\R$ is a sequence of distributions $(\mu_t)_{t\geq 0}$ that evolves according to the \red{continuity equation} $\partial_t \mu_t+\nabla \cdot \left(\mu_t v_t\right)=0$,
where the vector field $v_t = -\nabla_{W_2}\calF(\mu_t)$. 
Or equivalently, $\dot{x}_t = v_t(x_t)$ with $\mu_t = \mathrm{Law}(x_t)$. 
\end{defi}
\begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Euclidean Gradient Flow}
    \begin{itemize}
        \item State space: $\R^d$
        \item Objective $F: \R^d \to \R$.
        \item Update scheme: $\dot{x}_t = v_t(x_t)$, with $v_t = -\nabla F$. 
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Wasserstein Gradient Flow}
    \begin{itemize}
        \item State space: $\calP_2(\R^d)$
        \item Objective $\calF: \calP_2(\R^d) \to \R$:
        \item Update scheme $\dot{x}_t = v_t(x_t)$, with $v_t = -\nabla_{W_2} \calF(\mu_t)$. 
    \end{itemize}
\end{columns}
\end{frame}

\section{Examples of WGF}
\begin{frame}{Example 1: Langevin diffusion~\citep{jordan1998variational}}
    \begin{itemize}
        \item Given the target distribution $\pi\propto \exp(-V)$ with $V: \R^d\to\R$.
        \item The functional \textcolor{blue}{$\calF_\kl = \kl(\cdot \| \pi)$} and its Wasserstein gradient 
        \begin{talign*}
            \red{[\nabla_{W_2} \calF_\kl(\mu)](\cdot)} = \nabla V(\cdot) + \nabla \log \mu_t(\cdot) .
        \end{talign*}
        \item The Wasserstein gradient flow of $\calF_\kl$
        \begin{talign*}
            \partial_t \mu_t=\nabla \cdot \left(\mu_t \left( \nabla V + \nabla \log \mu_t\right) \right).
        \end{talign*}
        \item It is equivalent to the Fokker–Planck equation of the \green{Langevin diffusion}~\citep{sarkka2019applied}:
        \begin{talign*}
            \dd x_t = -\nabla V(x_t) \, \dd t + \sqrt{2} \, \dd W_t, \quad \mu_t = \mathrm{Law}(x_t).
        \end{talign*}
        \item A standard time-discretization (Euler--Maruyama scheme) is:
        \[
        x_{n+1} = x_n - \gamma \nabla V(x_n) + \sqrt{2\gamma} \, \xi_n, \quad \xi_n \sim \mathcal{N}(0, I_d).
        \]
    \end{itemize}
\end{frame}

\begin{frame}{Example 2: MMD gradient flow~\citep{arbel2019maximum}}
\begin{itemize}
    \item Given $M$ i.i.d samples $\{y_i\}_{i=1}^M$ from a target distribution $\pi$. 
    \item The functional \textcolor{blue}{$\calF_\mmd = \frac{1}{2}\mmd^2(\cdot \| \pi)$}  
    \item Its Wasserstein gradient 
    \begin{talign*}
        \red{[\nabla_{W_2} \calF_\mmd(\mu)](\cdot)} &= \nabla \left( \smallint k(x, \cdot) \; \dd \mu(x) - \smallint k(x, \cdot) \; \dd \pi(x) \right) \\
        &\approx \smallint \nabla_2 k(x, \cdot) \; \dd \mu(x) - \frac{1}{M}\sum_{i=1}^M \nabla_2 k(y_i, \cdot).
    \end{talign*}
    \item<2-> The Wasserstein gradient flow of $\calF_\mmd$,
    \begin{talign*}
        \partial_t \mu_t=\nabla \cdot \left(\mu_t \nabla_{W_2} \calF_\mmd(\mu_t) \right), \quad  \dd x_t = -[\nabla_{W_2} \calF_\mmd(\mu_t)](x_t) \, \dd t .
    \end{talign*}
    \item<2-> A standard time-discretization (Euler--Maruyama scheme) is:
    \[
    x_{n+1} = x_n - \gamma \nabla_{W_2} \calF_\mmd(\mu_n)(x_n) , \quad \text{Law}(x_n)=\mu_n. 
    \]
\end{itemize}
\end{frame}




\begin{frame}{Convergence}
    \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Wasserstein gradient flow of $\calF_\kl$}
    \begin{itemize}
        \item $\pi\propto\exp(-V)$ 
        \item Sampling 
        \item When $\pi\propto\exp(-V)$ is \red{strongly log-concave, i.e., $\bH V \geq \alpha\Id$}, then $\calF_\kl$ is \green{$\alpha$-convex}. 
        \item Convergence in discrete time 
        \begin{talign*}
            \calF_\kl\left(\mu_n\| \pi\right) \leq \exp(-\alpha \gamma n) \calF_\kl\left(\mu_n\| \pi\right) + \frac{\gamma n \beta^2}{\alpha}.
        \end{talign*}
        \item $\beta$ is the Lipschitz continuity of $V$. 
        \item It takes $\calO(\frac{1}{\alpha \delta} \log \frac{1}{\delta})$ to reach $\delta$ error.
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Wasserstein gradient flow of $\calF_\mmd$}
    \begin{itemize}
        \item $\{y_i\}_{i=1}^M$ i.i.d samples from $\pi$ 
        \item Generative modelling 
        \item When $k$ is bounded and has bounded derivatives, then $\calF_\mmd$ is \green{$M$-smooth and $-M$-convex}. 
        \item \textcolor{blue}{Non-}convergence in discrete time
        \begin{talign*}
            \calF_\mmd(\mu_n, \pi) \leq \frac{W_2^2(\mu_0,\pi)}{\gamma n} + \bar{K}.
        \end{talign*}
        \item $\bar{K}$ is a positive barrier term that does not vanish. 
        \item \blue{$\lim_{n\to\infty} \calF_\mmd(\mu_n, \pi) \neq 0$!}
    \end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Convexity and Smoothness}
\begin{exampleblock}{}
\centering
    \textbf{Question}: Can we find a new objective $\calF$ such that it enjoys \blue{convergence}, similar to $\calF_\kl$, in the \blue{generative modelling} setting where only samples are available?
\end{exampleblock}
\begin{prop}[MMD and $\chi^2$-divergence]
Suppose $\mu$ is absolutely continuous with respect to $\pi$, i.e., $\mu \ll \pi$. Then
\vspace{-10pt}
\begin{talign*}
    \operatorname{MMD}^2(\mu \| \pi)=\left\|T_\pi^{\frac{1}{2}} \left(\frac{\dd \mu}{\dd \pi}-1\right)\right\|_{L^2(\pi)}^2 \text { and } \chi^2(\mu \| \pi)=\left\|\frac{\dd \mu}{\dd \pi}-1 \right\|_{L^2(\pi)}^2 .
\end{talign*}
\vspace{-5pt}
Here, $T_\pi: L^2(\pi)\to L^2(\pi)$ is the kernel \red{integral operator} defined as
\begin{talign*}
    T_\pi f(\cdot)=\int k(x, \cdot) f(x) \; \dd \pi(x) . 
\end{talign*}
\end{prop}
\begin{itemize}
    \item Similar to $\calF_\kl$ flow, $\calF_{\chi^2}(\cdot)=\chi^2(\cdot\|\pi)$ flow enjoys convergence for Gaussian-like $\pi$~\citep{ohta2011displacement}. 
    \item An \green{interpolation} between $\mmd^2$ and $\chi^2$? 
\end{itemize}
\end{frame}


\begin{frame}{DrMMD: An interpolation of MMD and $\chi^2$-divergence}
\begin{defi}[De-regularized Maximum Mean Discrepancy (DrMMD)]
Suppose $\mu\ll\pi$ where $\mu, \pi \in \calP_2(\R^d)$. Then the (de)-regularized maximum mean discrepancy (DrMMD) between $\mu, \pi$ is defined as, for $\lambda>0$,
\vspace{-5pt}
\begin{talign*}
\operatorname{DrMMD}(\mu \| \pi)=(1+\lambda)\left\|\left(\left(T_\pi + \lambda \Id\right)^{-1} T_\pi\right)^{\frac{1}{2}} \left(\frac{\dd \mu}{\dd \pi}-1\right)\right\|_{L^2(\pi)}^2.
\end{talign*}
\end{defi}

\uncover<2->{%
\begin{prop}[Interpolation of $\mmd^2$ and $\chi^2$]
Suppose $k$ is bounded, continuous, and $c_0$-universal.
\begin{talign*}
    \lim _{\lambda \rightarrow 0} \operatorname{DrMMD}(\mu \| \pi)=\chi^2(\mu \| \pi), \quad\quad \lim _{\lambda \rightarrow \infty} \operatorname{DrMMD}(\mu \| \pi)=\operatorname{MMD}^2(\mu \| \pi) .
\end{talign*}
\end{prop}
\begin{itemize}
    \item<2-> Similar ideas based on spectral regularization have been used for kernel hypothesis testing~\citep{mika99fisher, eric2007testing, hagrass2022spectral}.
    \item<2-> Interpolation between MMD and general f-divergences~\citep{glaser2021kale,stein2026wasserstein}.
\end{itemize}
}
\end{frame}

\begin{frame}{DrMMD: An interpolation of MMD and $\chi^2$}

        \item Does $\calF_\dmmd$ flow admit \blue{finite sample implementation} of its Wasserstein gradient flow?

\begin{frame}{Kernel assumptions}
\begin{ass}[Growth I]
There exists a constant $M>0$ such that, for every $x\in\R^d$,
\begin{align*}
    \lvert k(x,x)\rvert
    &\leq M\left(1+\lvert x\rvert^4\right), \\
    \left\lvert \partial_{1,i}\partial_{2,i}k(x,x)\right\rvert
    &\leq M\left(1+\lvert x\rvert^2\right), \qquad i\in[d].
\end{align*}
\end{ass}

\uncover<2->{%
\begin{ass}[Local Boundedness]
For $R>0$, let $\mathcal K=B(0,R)\subset\R^d$. There exists a constant $B_R>0$ such that, for every $i,j\in[d]$,
\begin{align*}
    \sup_{x\in\mathcal K}
    \left\lVert \partial_{1,i}\partial_{1,j}k(x,\cdot)\right\rVert_{\mathcal H}
    \leq B_R.
\end{align*}
\end{ass}
\begin{itemize}
    \item<2-> Gaussian, Mat\'{e}rn, and inverse multiquadratic kernels.
\end{itemize}
}
\end{frame}

\begin{frame}{Finite-sample implementation}
\begin{prop}[Finite-sample estimate of the Wasserstein gradient of $\calF_\dmmd$]
The Wasserstein gradient of $\calF_\dmmd(\cdot) = \dmmd(\cdot\|\pi)$ at $\mu$ is \blue{$(1+\lambda)\nabla h_{\mu,\pi}(\cdot) : \R^d\to\R^d$}, where 
\begin{talign*}
    h_{\mu,\pi} = \left(T_\pi+\lambda \mathrm{I}\right)^{-1} T_\pi\left(\frac{\dd \mu}{\dd \pi}-1\right) .
\end{talign*}
\uncover<2->{%
Given empirical distributions $\hat{\mu} = \frac{1}{N} \sum_{i=1}^N x_i$ and $\hat{\pi}=\frac{1}{M} \sum_{i=1}^M y_i$. Given the Gram matrices $K_{x x} \in \R^{N \times N}$, $K_{yy} \in \R^{M \times M}$, $K_{xy} \in \R^{N \times M}$. 
\begin{talign*}
    h_{\hat{\mu}, \hat{\pi}}(\cdot) &= \frac{1}{N \lambda} k\left(\cdot, x_{1: N}\right) \one_N -\frac{1}{M\lambda} k\left(\cdot, y_{1: M}\right) \one_M - \frac{1}{M \lambda} k\left(\cdot, y_{1: M}\right)\left(M \lambda \mathrm{I}+K_{yy} \right)^{-1} K_{yx} \one_N \\ & + \frac{1}{M \lambda} k\left(\cdot, y_{1: M}\right)\left(M \lambda \mathrm{I}+K_{yy} \right)^{-1} K_{yy} \one_M . 
\end{talign*}
}
\end{prop}
\end{frame}



\begin{frame}{Convergence}
\begin{defi}[Poincar\'{e} inequality]
    We say that $\pi$ satisfies a \red{Poincaré inequality} with constant $C_{P}$ if for all $f , \nabla f \in L^2(\pi)$,
    \begin{talign*}
        \operatorname{Var}_\pi [f] \leq C_P \mathbb{E}_\pi \left[\|\nabla f\|^2\right] .
    \end{talign*} 
\end{defi}
\begin{itemize}
\item By ``Gaussian-like,'' we mean that $\pi$ satisfies a Poincar\'{e} inequality, which in particular implies sub-exponential tails~\citep{bakry2014analysis}.
\item For example, if $\pi=\calN(0,a \cdot \Id)$, then $C_P=a^{-1}$.
\end{itemize}
\begin{prop}[Convergence of $\calF_{\chi^2}$ flow]
Suppose that $\pi$ satisfies the  Poincar\'{e} inequality. Let $(\mu_t)_{t\geq 0}$ follow the $\calF_{\chi^2}$ flow. Then, for any $T \geq 0$,
$\kl\left(\mu_T \| \pi\right) \leq \exp \left(-2 C_P^{-1} T\right) \kl\left(\mu_0 \| \pi\right)$. 
\end{prop}
\end{frame}

\section{Convergence}
\begin{frame}{Convergence of DrMMD gradient flow}
\begin{thm}[Convergence of DrMMD gradient flow]
Let $(\mu_t)_{t\geq0}$ be the Wasserstein gradient flow of $\calF_{\dmmd}$. Suppose the target distribution $\pi$ \text satisfies a Poincaré inequality with constant $C_P$. Suppose that \orange{$\frac{\dd \mu_t}{\dd \pi}-1 \in \operatorname{Ran}(T_\pi^r)$ with $r>0$}. Then for any $T \geq 0$,
\begin{align*}
    \kl\left(\mu_T \| \pi\right) \leq \exp \left(-\frac{2(1+\lambda)}{C_P} T\right) \kl\left(\mu_0 \| \pi\right)+ \calO(\blue{\lambda^r} C_P) .
\end{align*}
\end{thm}
\begin{itemize}
    \item When $\lambda = 0$, it recovers the exponential convergence of $\chi^2$ flow.
    \begin{align*}
        \kl\left(\mu_T \| \pi\right) \leq \exp \left(-\frac{2}{C_P} T\right) \kl\left(\mu_0 \| \pi\right) .
    \end{align*}
    \item \blue{Larger $r$ means more regularity of the trajectory and thus smaller bias}. 
    \item Smaller Poincar\'{e} means faster convergence.
    \item For continuous time DrMMD flow, we would want $\lambda\to0$ for `convexity', however, that is not the case for discrete time flow. 
\end{itemize}
\end{frame}

    \caption{Visualization of $\operatorname{Ran}(T_\pi^r)$.}

\begin{itemize}
    \item \green{Large $r$ indicates larger smoothness.} 
    \item $\operatorname{Ran}(T_\pi^0) \cong L^2(\pi)$ and $\operatorname{Ran}(T_\pi^{\frac{1}{2}}) \cong \calH$.  
\end{itemize}
\end{frame}

\begin{frame}{DrMMD gradient descent}
\begin{itemize}
    \item DrMMD gradient flow (continuity equation)
    \begin{align*}
        \partial_t \mu_t+\nabla \cdot\left(\mu_t(1+\lambda) \nabla h_{\mu_t, \pi}\right)=0
    \end{align*}
    \item DrMMD gradient descent: for step size $\gamma > 0$, 
    \begin{align*}
        \mu_{n+1} = (\Id + \gamma (1+\lambda) \nabla h_{\mu_n, \pi})_\# \mu_n . 
    \end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Convergence of DrMMD gradient descent}
\begin{prop}[Descent lemma of DrMMD gradient descent]
Let $(\mu_n)_{n\in\N}$ be the Wasserstein gradient descent of $\calF_{\dmmd}$. 
Suppose $\pi\propto\exp(-V)$ with $\bH V \preceq \beta \mathrm{I}$. 
Suppose all assumptions in the proposition of the PL condition on $\calF_\dmmd$ hold. 
Suppose the step size $\gamma$ is small enough. 
\begin{talign*}
\kl\left(\mu_{n+1} \| \pi\right)-\kl\left(\mu_n \| \pi\right) & \leq-\frac{2}{C_P} \chi^2\left(\mu_n \| \pi\right) \gamma +\underbrace{\gamma \blue{\lambda^r}}_{\red{\text {Approximation error }} } + \underbrace{\gamma^2 \beta \chi^2\left(\mu_n \| \pi\right) \frac{1}{\blue{\lambda} }}_{\red{\text {Discretization error }} } 
\end{talign*}
\end{prop}
\begin{itemize}
    \item A \blue{trade-off} between the approximation error and the time-discretization error. 
    \item Optimal choice of \orange{adaptive} $\lambda_n$ at each iterate $n$: 
    \begin{talign*}
        \lambda_n \propto \chi^2\left(\mu_n \| \pi\right)^{\frac{1}{r+1}} 
    \end{talign*}
    \item At the start, we want \green{a larger $\lambda$} to have more smoothness; when closer to the convergence, we want \green{a smaller $\lambda$} to operate in the $\chi^2$ regime to better catch the difference.  
\end{itemize}
\end{frame}

\begin{frame}{Convergence of DrMMD gradient descent}
\begin{thm}[Convergence of DrMMD gradient descent]
Let $(\mu_n)_{n\in\N}$ be the Wasserstein gradient descent of $\calF_{\dmmd}$. Suppose all conditions from the descent lemma hold. Then, for any $n_{\max}\in\N$, 
\begin{align*}
    \kl\left(\mu_{n_{\max}} \| \pi\right) & \leq \exp \left(-\frac{ 2 n_{\max } \gamma}{C_P}\right) \kl\left(\mu_0 \| \pi\right) + \calO( \blue{\gamma^{\frac{r}{r+1}}} C_P \beta^{\frac{r}{r+1}} ). 
\end{align*}
\end{thm}
\begin{itemize}
    \item To reach error $\kl\left(\mu_{n_{\max}} \| \pi\right)\leq \delta$, it takes $\mathcal{O}(\blue{(\frac{1}{\delta})^{\frac{r+1}{r}}} \log \frac{1}{\delta})$ iterations.
    \item By comparison, for Langevin Monte Carlo, it takes $\mathcal{O}(\blue{\frac{1}{\delta}} \log \frac{1}{\delta})$~\citep{chewi2024analysis}. 
    \item DrMMD gradient descent takes more iterations due to the additional approximation error $\calO(\lambda^r)$, but it \blue{operates without the knowledge of potential $V$} and only requires $\Sigma_\pi=\int k(x,\cdot) \otimes k(x,\cdot) \; \dd \pi$ and the embedding $\int k(x,\cdot) \; \dd \pi$.  
\end{itemize}
\end{frame}

\section{Conclusions}
\begin{frame}{Conclusions}
\begin{itemize}
    \item We propose DrMMD gradient flow as an interpolation of MMD gradient flow and $\chi^2$ gradient flow.
    \item DrMMD gradient flow / descent has \green{global convergence} results, compared to MMD flow, under an adaptive regularization parameter $\lambda$. 
    \item This justifies the application of \blue{adaptive kernels} in recent MMD flow (MMD GAN) papers that achieve SOTA empirical performances.
    \item More in the paper \url{https://arxiv.org/pdf/2409.14980}.
    \begin{itemize}
        \item Empirical results on synthetic datasets.
        \item An example of DrMMD flow with a Gaussian target distribution $\pi$ which satisfies all conditions in the theorems. 
        \item Finite-particle convergence results with propagation of chaos bound. 
    \end{itemize}
\end{itemize}
\end{frame}
